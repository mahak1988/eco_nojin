function WaterSurface({ levelNorm }: { levelNorm: number }) {







  const ref = useRef<THREE.Mesh>(null!);







  const mat = useRef<THREE.MeshStandardMaterial>(null!);







  useFrame(({ clock }) => {







    const t = clock.getElapsedTime();







    if (mat.current) mat.current.opacity = 0.5 + 0.1 * Math.sin(t * 0.9);







    if (ref.current) ref.current.position.y = levelNorm * HEIGHT_SCALE + 0.4 * Math.sin(t * 0.5);







  });







  return (







    <mesh ref={ref} rotation={[-Math.PI / 2, 0, 0]}>







      <planeGeometry args={[WORLD_SIZE * 0.98, WORLD_SIZE * 0.98]} />







      <meshStandardMaterial ref={mat} color="#2f6f9f" transparent opacity={0.55} metalness={0.35} roughness={0.12} />







    </mesh>







  );







}
