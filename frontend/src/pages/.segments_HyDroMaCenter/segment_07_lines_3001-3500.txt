function CameraTour({ active }: { active: boolean }) {







  const { camera } = useThree();







  const t = useRef(0);







  useFrame((_, dt) => {







    if (!active) return;







    t.current += dt;







    const ang = t.current * 0.18;







    const radius = 1150 - 350 * (0.5 + 0.5 * Math.sin(t.current * 0.1));







    const height = 520 - 240 * (0.5 + 0.5 * Math.cos(t.current * 0.13));







    camera.position.set(Math.cos(ang) * radius, height, Math.sin(ang) * radius);







    camera.lookAt(0, 30, 0);







  });







  return null;







}
